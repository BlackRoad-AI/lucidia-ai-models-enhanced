Quickstart

Up and running in 60 seconds.

Install the SDK, point it at a model, and start quantizing. No GPU required.

⬡ npm install Read the docs →
python 
from src.ai_models_enhanced import (
    EnhancedModelPipeline,
    QuantizationConfig,
    LoRAAdapter, FineTuneJob,
)

pipeline = EnhancedModelPipeline()

# Step 1 — Quantize to AWQ 4-bit
q = pipeline.quantize_model(QuantizationConfig(
    model_id="llama3-8b",
    method="awq",
    bits=4,
))
print(f"Reduced by {q.size_reduction_pct}%")

# Step 2 — Attach LoRA adapter
adapter = pipeline.add_lora(LoRAAdapter(
    model_id="llama3-8b",
    name="code-instruct-v1",
    rank=16, alpha=32.0,
))

# Step 3 — Fine-tune and track
job = pipeline.track_finetune(FineTuneJob(
    model_id="llama3-8b",
    dataset="alpaca-cleaned-52k",
    epochs=3, learning_rate=2e-4,
))
print(f"Final loss: {job.train_loss:.4f}")

pipeline.close()
